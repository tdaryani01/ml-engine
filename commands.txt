import yappi
from benchmarks.benchmark_cnn import _benchmark_common_from_config, run_pytorch_benchmark

c = _benchmark_common_from_config(None)
yappi.set_clock_type("wall")
yappi.clear_stats()
yappi.start(builtins=False)
try:
    run_pytorch_benchmark(
        X_train=c["X_train"],
        y_train_classes=c["y_train_classes"],
        X_val=c["X_val"],
        y_val_classes=c["y_val_classes"],
        num_classes=c["num_classes"],
        batch_size=c["batch_size"],
        epochs=c["epochs"],
        lr_init=c["lr_init"],
        lam_l2=c["lam_l2"],
        early_stopping_enabled=c["early_stopping_enabled"],
        patience=c["patience"],
        min_delta=c["min_delta"],
        cnn_dict=c["cnn_dict"],
        num_threads=c["num_threads"],
    )
finally:
    yappi.stop()

out = "yappi_pytorch_pad2.pstat"
yappi.get_func_stats().save(out, type="pstat")
print(f"saved {out}")
PY
